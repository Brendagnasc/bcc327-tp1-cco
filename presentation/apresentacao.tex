\section{Contexto e Arquitetura}
% =========================================================

\begin{frame}{Do TP1 ao TP2}
\begin{columns}[c]
\column{0.40\textwidth}
\includegraphics[width=\linewidth]{m1_tubos}

\column{0.57\textwidth}
\begin{block}{Problema}
\small
Uma árvore 3D com 511 vasos tem ramos sobrepostos em vários níveis de profundidade. Para ser interpretável, a imagem precisa de:
\end{block}
\small
\begin{itemize}
  \item[\checkitem] \textbf{oclusão correta} entre os vasos
  \item[\checkitem] \textbf{iluminação} que revele a forma cilíndrica
  \item[\checkitem] \textbf{transparência} para ver o interior
  \item[\checkitem] \textbf{interação}: consultar e isolar vasos
\end{itemize}
\vspace{0.1cm}
No TP1, iluminação e visibilidade eram \alert{caixa-preta} do OpenGL.
\end{columns}
\end{frame}

\fig{0.19}{m2_zbuffer_culling_proprio}{Z + culling próprio}\hfill
\fig{0.19}{m2_profundidade}{profundidade}

\vspace{0.2cm}
\small
\begin{columns}[t]
\column{0.50\textwidth}
\begin{block}{Culling próprio}
\scriptsize
descarta se $\mathbf{N}\cdot\mathbf{V} \le 0$, com $\mathbf{V} = -\text{centroide}$ (perspectiva)\\
\alert{4\,546 de 9\,192 triângulos} (49,5\%), imagem idêntica
\end{block}
\column{0.47\textwidth}
\begin{block}{Conclusão}
\scriptsize
faces de trás em \textcolor{red}{vermelho}: culling sozinho \alert{não} resolve a oclusão \emph{entre} vasos; só o Z-Buffer
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Culling: próprio $\times$ OpenGL e o Z-Buffer}
\begin{columns}[t]
\column{0.48\textwidth}
\begin{block}{Dois critérios, mesmo resultado}
\small
\checkitem\ \textbf{próprio}: geometria antes da projeção ($\mathbf{N}\cdot\mathbf{V}$)\\
\checkitem\ \textbf{OpenGL}: ordem dos vértices na tela (anti-horário = frente)\\
\checkitem\ exige faces orientadas: $(\mathbf{v}_1 - \mathbf{v}_0)\times(\mathbf{v}_2 - \mathbf{v}_0)$ para fora, garantido em \texttt{geometria.py}
\end{block}
\column{0.48\textwidth}
\begin{block}{Profundidade não linear}
\small
\[ z = \frac{2nf}{(f+n) - (2d-1)(f-n)} \]
lido com \texttt{glReadPixels}, linearizado e normalizado\\
($n = 0{,}01$, $f = 100$): a precisão se concentra perto do plano near
\end{block}
\end{columns}
\end{frame}

\begin{block}{Modelo de iluminação (espaço da câmera)}
\centering
$I = k_a\,\mathbf{c} + I_L\left[k_d\,\mathbf{c}\,\max(\mathbf{N}\cdot\mathbf{L},0) + k_s\max(\mathbf{N}\cdot\mathbf{H},0)^{\alpha}\right]$, \quad $\mathbf{H} = \dfrac{\mathbf{L}+\mathbf{V}}{\|\mathbf{L}+\mathbf{V}\|}$\\[0.1cm]
{\small $k_a = 0{,}25$, $k_d = 0{,}75$, $k_s = 0{,}6$, $\alpha = 40$ | normais por $(M_{3\times3}^{-1})^T$}
\end{block}
\end{frame}

\checkitem\ Gouraud $\approx$ Phong nos vasos finos; nos grossos o Gouraud borra o brilho ao longo dos triângulos longos

\section{Técnicas Baseadas em Raios}
% =========================================================

\begin{frame}{Seleção de vasos por Ray Casting}
\begin{columns}[c]
\column{0.38\textwidth}
\includegraphics[width=\linewidth]{m6_selecao}\\
{\scriptsize vaso 9 | geração 2 | raio 0,893\,mm | $t = 0{,}0202$}

\column{0.59\textwidth}
\small
\checkitem\ pixel $\to$ NDC $\to$ $(PV)^{-1}$: raio $\mathbf{O} + t\mathbf{D}$\\
\checkitem\ levado ao espaço do modelo por $M^{-1}$\\
\checkitem\ vaso = cilindro + 2 esferas; escolhe o \alert{menor $t > 0$}
\begin{block}{Raio--cilindro}
\scriptsize
$k_2 t^2 + k_1 t + k_0 = 0$, com $\mathbf{m} = \mathbf{O} - \mathbf{A}$:\\
$k_2 = \mathbf{D}\cdot\mathbf{D} - (\mathbf{D}\cdot\mathbf{a})^2$, \
$k_1 = 2[\mathbf{m}\cdot\mathbf{D} - (\mathbf{m}\cdot\mathbf{a})(\mathbf{D}\cdot\mathbf{a})]$\\
$k_0 = \mathbf{m}\cdot\mathbf{m} - (\mathbf{m}\cdot\mathbf{a})^2 - r^2$, \ válido se $0 \le s \le h$
\end{block}
\scriptsize vetorizado: $\approx$ 0,6\,ms para os 1\,023 vasos do Nterm512
\end{columns}
\end{frame}

\begin{frame}{Discussão}
\begin{columns}[t]
\column{0.48\textwidth}
\begin{block}{Comparações}
\small
\checkitem\ culling = otimização; Z-Buffer = visibilidade\\
\checkitem\ Phong: brilho correto, custo absorvido pela GPU\\
\checkitem\ rasterização: interativa, sem sombras nem reflexões\\
\checkitem\ Ray Tracing: sombras e reflexões naturais, mas custo pixels $\times$ objetos $\times$ raios
\end{block}
\column{0.48\textwidth}
\begin{block}{Limitações}
\small
\checkitem\ Painter por centroide falha em interpenetrações (junções)\\
\checkitem\ ordenação refeita a cada quadro em Python\\
\checkitem\ Ray Tracer numa cena simplificada, sem antialiasing\\
\checkitem\ Nterm512 lento no WSL (renderização por software)
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Conclusões}
\begin{itemize}
  \item[\checkitem] pipeline programável com iluminação e sombreamento próprios
  \item[\checkitem] visibilidade estudada com culling próprio, culling do OpenGL e Z-Buffer
  \item[\checkitem] transparência ordenada e textura com filtragem bilinear própria
  \item[\checkitem] raios na interação (seleção) e na síntese de imagens (Ray Tracer)
  \item[\checkitem] métricas e filtro: 1/4 dos vasos concentra mais da metade do volume
\end{itemize}
\vspace{0.2cm}
\begin{block}{Trabalhos futuros}
Ray Tracer na árvore completa com BVH | transparência independente de ordem (OIT) | sombras por \emph{shadow mapping}
\end{block}
\end{frame}
